    import { xySlider } from "@mootari/2d-slider"

// ========= Controls =========
//| panel: input

// Sample size
viewof n = Inputs.range([20, 100], { value: 50, step: 5, label: md`${tex`n`}` })

// True parameters used to SIMULATE the data
viewof trueMu = Inputs.range([-2, 2], { value: 0.5, step: 0.1, label: md`${tex`\mu_{\text{true}}`}` })
viewof trueSigma = Inputs.range([0.3, 3], { value: 1.2, step: 0.1, label: md`${tex`\sigma_{\text{true}}`}` })

// Grid controls
viewof muHalfWidth = Inputs.range([1, 5], { value: 2.5, step: 0.1, label: md`${tex`\text{μ-range half-width}`}` })
viewof sig2Max = Inputs.range([0.2, 6], { value: 3, step: 0.1, label: md`${tex`\text{σ² max}`}` })
viewof gridN = Inputs.range([25, 121], { value: 81, step: 8, label: md`${tex`\text{grid size}`}` })

// ========= Data & stats =========

// Simulate iid Normal data via Box–Muller
X = {
  const m = trueMu, s = trueSigma;
  const out = new Array(n);
  for (let i = 0; i < n; i += 2) {
    const u = Math.random(), v = Math.random();
    const R = Math.sqrt(-2 * Math.log(u)), T = 2 * Math.PI * v;
    const z0 = R * Math.cos(T), z1 = R * Math.sin(T);
    out[i]   = m + s * z0;
    if (i + 1 < n) out[i + 1] = m + s * z1;
  }
  return out;
}

// Sufficient stats
xbar = d3.mean(X)
S = d3.sum(X, x => (x - xbar) ** 2)       // Σ (x_i - x̄)²

// Log-likelihood ℓ(μ,σ²) (exact, not just up to constant)
logLik = (mu, sigma2) => {
  const ss = d3.sum(X, x => (x - mu) ** 2);
  return -0.5 * n * Math.log(2 * Math.PI * sigma2) - 0.5 * ss / sigma2;
}

// MLEs: μ̂ = x̄,  σ̂² = (1/n) Σ (x_i - x̄)²
muhat = xbar
sigma2hat = S / n

// ========= Contour grid =========

muGrid = d3.ticks(muhat - muHalfWidth, muhat + muHalfWidth, gridN - 1)
sig2Grid = d3.ticks(Math.max(0.05, sigma2hat / 20), sig2Max, gridN - 1).filter(s => s > 0)

// Grid for Plot: [{mu, sig2, ll}]
LLgrid = {
  const out = [];
  for (const mu of muGrid) {
    for (const s2 of sig2Grid) {
      out.push({ mu, sig2: s2, ll: logLik(mu, s2) });
    }
  }
  return out;
}

// ========= 2D probe (slider) =========

// NOTE: xySlider may return either [x, y] or {x, y}. We'll read both safely.
viewof probe = xySlider({
  width: 360, height: 360,
  x: { min: muGrid[0],   max: muGrid.at(-1), value: muhat,     label: md`${tex`\mu`}` },
  y: { min: sig2Grid[0], max: sig2Grid.at(-1), value: sigma2hat, label: md`${tex`\sigma^2`}` },
  grid: false
})

// Robust extraction with fallbacks to MLEs (prevents .toFixed errors)
probeMu = (Array.isArray(probe) ? probe[0] : probe?.x) ?? muhat
probeSig2 = (Array.isArray(probe) ? probe[1] : probe?.y) ?? sigma2hat
probeLL = logLik(probeMu, probeSig2)

// ========= Readout =========
md`
${tex`\textbf{Probe: } \mu = ${probeMu.toFixed(2)},\ \sigma^2 = ${probeSig2.toFixed(3)}\quad \Rightarrow\quad \ell = ${probeLL.toFixed(2)}`}
${tex`\textbf{MLE: } \hat\mu = ${muhat.toFixed(2)},\ \hat\sigma^2 = ${sigma2hat.toFixed(3)}`}
`
    md`${tex`\textbf{Log-likelihood contours } \ell(\mu,\sigma^2)\ \text{for Normal}(X_i \stackrel{iid}{\sim} \mathcal N(\mu,\sigma^2))`}`
    Plot.plot({

  marginLeft: 48,

    // Heatmap background (helps orientation)
    Plot.cell(LLgrid, { x: "mu", y: "sig2", fill: "ll", tip: true }),

    // Contours of ℓ(μ,σ²)
    Plot.contour(LLgrid, {
      x: "mu",
      y: "sig2",
      z: "ll",
      thresholds: 10,
      stroke: "currentColor",
      fill: "none"
    }),

    // MLE marker
    Plot.dot([{ mu: muhat, sig2: sigma2hat }], {

    }),

    // Crosshairs at the probe
    Plot.ruleX([probeMu]),
    Plot.ruleY([probeSig2])
  ]
})
