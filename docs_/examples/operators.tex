\subsection{\texorpdfstring{\texttt{Partial}
Operator}{Partial Operator}}\label{partial-operator}

\subsubsection{What it is}\label{what-it-is}

\begin{itemize}
\tightlist
\item
  \textbf{Operator:} \(\mathrm{Partial}(j)\) maps \(f\) to
  \(\partial_{x_j} f\).
\item
  \textbf{As a probe:}
  \(F^{(\partial)} = \text{Probe}((\mathrm{Partial}(j),),\, \mathrm{Eval}(X))\).
\end{itemize}

\subsubsection{Meaning}\label{meaning}

\[
F^{(\partial)}(f) \;=\; \Big(\tfrac{\partial}{\partial x_j} f(x_i)\Big)_{i=1}^n \in \mathbb{R}^{n\times m}.
\]

\subsubsection{Cross-covariances via operator
calculus}\label{cross-covariances-via-operator-calculus}

For a scalar base kernel \(k\),

\[
\begin{aligned}
\operatorname{cov}\!\big[\partial_{x_j} f(x),\, f(y)\big] &= \partial_{x_j} k(x,y),\\
\operatorname{cov}\!\big[\partial_{x_j} f(x),\, \partial_{y_\ell} f(y)\big] &= \partial_{x_j}\partial_{y_\ell} k(x,y).
\end{aligned}
\]

Thus the Gram for two probes is obtained by applying operators to \(k\)
\textbf{then} evaluating:

\[
K_{F^{(\partial)},\,G} \;=\; (\mathrm{Eval}\otimes \mathrm{Eval})\big[(\mathrm{Partial}(j)\otimes \mathcal{O}_G)\,k\big].
\]

\subsubsection{Conditioning on derivative
data}\label{conditioning-on-derivative-data}

Given noisy derivative observations
\(y^{(\partial)} \approx \partial_{x_j} f(X)\),

\[
\begin{aligned}
m_{\text{post}}(x_\star)
&= m(x_\star) + K_{x_\star,\,\partial X}\,\big(K_{\partial X,\partial X}+\Sigma_\varepsilon\big)^{-1}\!\big(y^{(\partial)} - m_{\partial}(X)\big),\\[4pt]
k_{\text{post}}(x_\star,x_\star)
&= k(x_\star,x_\star) - K_{x_\star,\,\partial X}\,\big(K_{\partial X,\partial X}+\Sigma_\varepsilon\big)^{-1}\,K_{\partial X,\,x_\star},
\end{aligned}
\]

with blocks like
\(K_{x_\star,\,\partial X} = \partial_{x} k(x_\star, X)\),
\(K_{\partial X,\partial X}=\partial_x\partial_y k(X,X)\).

\subsection{\texorpdfstring{\texttt{Grad}
Operator}{Grad Operator}}\label{grad-operator}

We may want to condition a Gaussian process not only on \textbf{function
values} but also on their \textbf{gradients}.\\
This is made possible by introducing the \textbf{gradient operator}:

\[
(\nabla f)(x) = \left( \frac{\partial f}{\partial x_1}(x), \ldots, 
                     \frac{\partial f}{\partial x_d}(x) \right)^\top .
\]

This is implemented in \texttt{GaussED} as the \texttt{Grad} operator.

\subsubsection{\texorpdfstring{\textbf{Gradient conditioned on
evaluations}}{Gradient conditioned on evaluations}}\label{gradient-conditioned-on-evaluations}

\begin{itemize}
\tightlist
\item
  Data: \(f(X)\) observed.\\
\item
  Posterior: joint distribution of \(\{ f(X), \nabla f(X_*) \}\).\\
\item
  Effect: gradients predicted consistently with observed values.
\end{itemize}

\subsection{\texorpdfstring{\texttt{Linearise}
Operator}{Linearise Operator}}\label{linearise-operator}

Suppose we have \(f: \mathcal{X} \rightarrow \mathbb{R}^D\) and a
function \(g:\mathbb{R}^D\rightarrow \mathbb{R}^D\). Then \(g \circ f\)
is Gaussian if and only if \(g\) is linear.

\textbf{Approximate Solution:} Linearise
\(g(y) \approx g(y_0) + J_g(y_0)(y - y_0)\). About a reference \(f_0\),
we have:

\[ 
g(f(x)) \approx g(f_0(x)) + J_g(f_0(x))(f(x) - f_0(x)). 
\]

This is used in Warped-Sequential Active Bayesian integration (WSABI) of
Gunter et al.~(2019).

\textbf{Motivation}: Want to normalise posteriors (some important prior
information is that posterior density is non-negative!)

In the following we linearise based upon \(g(y) = \alpha + y^2/2\).
